using System;
using System.IO;
using System.Linq;
using System.Collections.Generic;
using System.Diagnostics;
using System.Globalization;
using System.Security.Cryptography;
using System.Text.Json;
using RailwayRenaissance.Cab.Simulation;
using RailwayRenaissance.Cab.Research;

static class Program
{
    static string F(double x)=>x.ToString("R",CultureInfo.InvariantCulture);
    static AtoObservation Observe(SimulationState s)=>new AtoObservation{Tick=s.Tick,Position=s.PositionM,Speed=s.SpeedMps,Acceleration=s.Acceleration,PipeKpa=s.PipeKpa,ControlKpa=s.ControlKpa,CylinderKpa=s.TrainCylinderKpa};
    static void Require(bool ok,string why){if(!ok)throw new Exception(why);}
    static SimulationState Ready(int seed)
    {
        var m=new CabSimulation(seed);var u=new SimulationInput{Master=true,FuelPump=true,EngineRun=true,Brake=1};
        for(int i=0;i<15000;i++)m.Step(u);
        Require(m.State.Running&&m.State.PipeKpa>490,"Startup preparation failed");return m.State.Copy();
    }
    sealed class Result
    {
        public string method {get;set;} public int seed {get;set;}
        public double mass {get;set;} public double grip {get;set;} public double grade {get;set;}
        public bool stopped {get;set;} public bool reachedTolerance {get;set;}
        public double errorM {get;set;} public double seconds {get;set;} public int switches {get;set;}
        public double decisionMaxMs {get;set;} public double decisionP95Ms {get;set;}
        public string finalState {get;set;}
        public string initialState {get;set;} public string configuration {get;set;}
        public double penalty {get;set;} public int dwellTicks {get;set;}
        public double targetDistance {get;set;} public double targetSpeed {get;set;}
        public string stressCase {get;set;} public int observationDelayTicks {get;set;}
        public double brakeForceScale {get;set;} public bool gripDrop {get;set;}
        public double fuelUsedL {get;set;} public double exhaustAirKpaL {get;set;} public double compressorAirKpaL {get;set;}
        public double peakJerk {get;set;} public double movingJerkRms {get;set;} public int movingJerkSamples {get;set;}
        public double peakSpeedMps {get;set;} public double preBrakeSpeedRmse {get;set;} public double preBrakeSeconds {get;set;}
    }
    static Result Run(AtoMethod method,int seed,double mass,double grip,double grade,string trace=null,int frame=1,double penalty=.15,int dwellTicks=0,double distance=900,double limit=16,int delayTicks=0,double brakeScale=1,bool gripDrop=false,string stressCase="none")
    {
        var start=Ready(seed);start.TrainMassKg*=mass;
        if(delayTicks<0||delayTicks>50||brakeScale<=0)throw new ArgumentException("Invalid stress profile");
        var config=new SimulationConfig{ServiceBrakeN=180000*brakeScale};
        var m=new CabSimulation(seed,config);m.Restore(start);
        double target=start.PositionM+distance;var controller=new CabAtoController(method,target,limit,180,penalty,dwellTicks);
        var u=new SimulationInput{Master=true,FuelPump=true,EngineRun=true,ParkingBrake=false,Direction=1,Brake=1,RailGrip=grip,GradePermille=grade};
        using var writer=trace==null?null:new StreamWriter(trace);
        writer?.WriteLine("tick,time_s,position_m,target_m,speed_mps,pipe_kpa,cylinder_kpa,throttle,brake,requested_throttle,estimated_decel,plans,acceleration_mps2,grip,fuel_used_l,exhaust_air_kpal");
        int stopped=0,switches=0,last=1,ticks=0;bool didStop=false;var times=new List<double>();
        var history=new Queue<AtoObservation>();
        double peakJerk=0,jerkSq=0,peakSpeed=0,speedSq=0;int jerkSamples=0,speedSamples=0;
        double startFuel=m.State.FuelUsedL,startExhaust=m.State.ExhaustAirKpaL,startCompressor=m.State.CompressorAirKpaL;
        while(ticks<15000&&!didStop)
        {
            for(int f=0;f<frame&&ticks<15000&&!didStop;f++)
            {
                int requested=u.Throttle;
                history.Enqueue(Observe(m.State));
                while(history.Count>delayTicks+1)history.Dequeue();
                if(ticks%5==0)
                {
                    var observed=history.Peek();observed.Tick=m.State.Tick;
                    var watch=Stopwatch.StartNew();var command=controller.Decide(observed);watch.Stop();times.Add(watch.Elapsed.TotalMilliseconds);
                    requested=command.Throttle;u.Brake=command.Brake;
                    // Match the production facade's traction availability guards.
                    u.Throttle=requested>0&&(m.State.PipeKpa<470||m.State.MainKpa<600||m.State.TrainCylinderKpa>=20||u.Brake>=2)?0:requested;
                    if(u.Brake!=last)switches++;last=u.Brake;
                }
                u.RailGrip=gripDrop&&m.State.PositionM-start.PositionM>=450?grip*.6:grip;
                double previousSpeed=m.State.SpeedMps,previousAcceleration=m.State.Acceleration;
                m.Step(u);ticks++;
                double jerk=(m.State.Acceleration-previousAcceleration)/.02;
                peakJerk=Math.Max(peakJerk,Math.Abs(jerk));peakSpeed=Math.Max(peakSpeed,Math.Abs(m.State.SpeedMps));
                if(Math.Abs(previousSpeed)>.5&&Math.Abs(m.State.SpeedMps)>.5){jerkSq+=jerk*jerk;jerkSamples++;}
                if(!controller.BrakingPhase){speedSq+=Math.Pow(m.State.SpeedMps-limit,2);speedSamples++;}
                if(ticks%5==0)writer?.WriteLine(string.Join(",",m.State.Tick,F(m.State.Time),F(m.State.PositionM),F(target),F(m.State.SpeedMps),F(m.State.PipeKpa),F(m.State.TrainCylinderKpa),u.Throttle,u.Brake,requested,F(controller.EstimatedDeceleration),controller.PlansLastUpdate,F(m.State.Acceleration),F(u.RailGrip),F(m.State.FuelUsedL-startFuel),F(m.State.ExhaustAirKpaL-startExhaust)));
                stopped=controller.BrakingPhase&&Math.Abs(m.State.SpeedMps)<.01?stopped+1:0;
                didStop=stopped>=100;
            }
        }
        times.Sort();double error=m.State.PositionM-target;
        return new Result{method=method.ToString(),seed=seed,mass=mass,grip=grip,grade=grade,penalty=penalty,dwellTicks=dwellTicks,targetDistance=distance,targetSpeed=limit,stressCase=stressCase,observationDelayTicks=delayTicks,brakeForceScale=brakeScale,gripDrop=gripDrop,fuelUsedL=m.State.FuelUsedL-startFuel,exhaustAirKpaL=m.State.ExhaustAirKpaL-startExhaust,compressorAirKpaL=m.State.CompressorAirKpaL-startCompressor,peakJerk=peakJerk,movingJerkRms=jerkSamples>0?Math.Sqrt(jerkSq/jerkSamples):0,movingJerkSamples=jerkSamples,peakSpeedMps=peakSpeed,preBrakeSpeedRmse=speedSamples>0?Math.Sqrt(speedSq/speedSamples):0,preBrakeSeconds=speedSamples*.02,stopped=didStop,reachedTolerance=didStop&&Math.Abs(error)<=5,errorM=error,seconds=ticks*.02,switches=switches,decisionMaxMs=times.Max(),decisionP95Ms=times[(int)((times.Count-1)*.95)],initialState=SimulationCodec.Encode(start),configuration=SimulationCodec.Encode(m.Config),finalState=SimulationCodec.Encode(m.State)};
    }
    static string Hash(string p)=>Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(p)));
    static void Main(string[] args)
    {
        string dir=args.Length>0?args[0]:"research/ato/results/development-"+DateTime.UtcNow.ToString("yyyyMMdd-HHmmss");
        if(File.Exists(Path.Combine(dir,"results.json")))throw new IOException("Refusing to overwrite an existing experiment");
        Directory.CreateDirectory(dir);
        var results=new List<Result>();
        foreach(AtoMethod method in Enum.GetValues<AtoMethod>())
        {
            var a=Run(method,1000,1,1,0,Path.Combine(dir,method+".csv"));
            var b=Run(method,1000,1,1,0,null,7);
            Require(a.finalState==b.finalState,"Frame-group replay mismatch: "+method);
            results.Add(a);
            results.Add(Run(method,1001,1.2,.3,-10));
        }
        if(args.Length>1&&args[1]=="--grid")
        {
            results.Clear();
            foreach(AtoMethod method in Enum.GetValues<AtoMethod>())foreach(int seed in new[]{1000,1001})
            foreach(double mass in new[]{.8,1.0,1.2})foreach(double grip in new[]{.3,.55,1.0})foreach(double grade in new[]{-10.0,0,10.0})
                results.Add(Run(method,seed,mass,grip,grade,Path.Combine(dir,$"{method}-{seed}-{F(mass)}-{F(grip)}-{F(grade)}.csv")));
        }
        if(args.Length>1&&args[1]=="--tune")
        {
            results.Clear();
            foreach(double penalty in new[]{.15,1.0,4.0})foreach(int dwell in new[]{0,15,25})
            foreach(AtoMethod method in new[]{AtoMethod.EarlyPredictive,AtoMethod.AdaptivePredictive})foreach(int seed in new[]{1000,1001})
            foreach(double mass in new[]{.8,1.0,1.2})foreach(double grip in new[]{.3,.55,1.0})foreach(double grade in new[]{-10.0,0,10.0})
                results.Add(Run(method,seed,mass,grip,grade,null,1,penalty,dwell));
        }
        bool heldout=args.Length>1&&args[1]=="--holdout";
        if(heldout)
        {
            results.Clear();
            using var protocol=JsonDocument.Parse(File.ReadAllText("research/ato/heldout-protocol-v1.json"));
            var p=protocol.RootElement;
            foreach(string name in p.GetProperty("methods").EnumerateArray().Select(v=>v.GetString()))
            foreach(int seed in p.GetProperty("seeds").EnumerateArray().Select(v=>v.GetInt32()))
            foreach(double mass in p.GetProperty("mass").EnumerateArray().Select(v=>v.GetDouble()))
            foreach(double grip in p.GetProperty("grip").EnumerateArray().Select(v=>v.GetDouble()))
            foreach(double grade in p.GetProperty("grade").EnumerateArray().Select(v=>v.GetDouble()))
            foreach(double distance in p.GetProperty("distance").EnumerateArray().Select(v=>v.GetDouble()))
            foreach(double speed in p.GetProperty("speed").EnumerateArray().Select(v=>v.GetDouble()))
            {
                var method=Enum.Parse<AtoMethod>(name);
                var dwellValues=method==AtoMethod.EarlyPredictive||method==AtoMethod.AdaptivePredictive?new[]{0,p.GetProperty("dwellTicks").GetInt32()}:new[]{p.GetProperty("dwellTicks").GetInt32()};
                foreach(int dwell in dwellValues)
                {
                    string path=Path.Combine(dir,$"{name}-{seed}-{F(mass)}-{F(grip)}-{F(grade)}-{F(distance)}-{F(speed)}-{dwell}.csv");
                    results.Add(Run(method,seed,mass,grip,grade,path,1,p.GetProperty("penalty").GetDouble(),dwell,distance,speed));
                }
            }
            foreach(var method in new[]{AtoMethod.EarlyPredictive,AtoMethod.AdaptivePredictive})
            {
                var a=Run(method,1000,1,1,0,null,1,.15,25);
                var b=Run(method,1000,1,1,0,null,7,.15,25);
                Require(a.finalState==b.finalState,"Dwell frame replay mismatch");
            }
            File.Copy("research/ato/heldout-protocol-v1.json",Path.Combine(dir,"protocol.json"));
        }
        bool stress=args.Length>1&&args[1]=="--stress";
        if(stress)
        {
            results.Clear();
            using var protocol=JsonDocument.Parse(File.ReadAllText("research/ato/stress-protocol-v1.json"));
            var p=protocol.RootElement;
            Require(Hash("Assets/Scripts/Cab/Research/CabAtoController.cs")==p.GetProperty("controllerHash").GetString(),"Stress controller differs from frozen protocol");
            foreach(string name in p.GetProperty("methods").EnumerateArray().Select(v=>v.GetString()))
            foreach(int seed in p.GetProperty("seeds").EnumerateArray().Select(v=>v.GetInt32()))
            foreach(int dwell in p.GetProperty("dwellTicks").EnumerateArray().Select(v=>v.GetInt32()))
            foreach(var profile in p.GetProperty("cases").EnumerateArray())
            {
                string scenario=profile.GetProperty("name").GetString();
                results.Add(Run(Enum.Parse<AtoMethod>(name),seed,p.GetProperty("mass").GetDouble(),p.GetProperty("grip").GetDouble(),p.GetProperty("grade").GetDouble(),Path.Combine(dir,$"{name}-{seed}-{dwell}-{scenario}.csv"),1,p.GetProperty("switchPenalty").GetDouble(),dwell,p.GetProperty("distance").GetDouble(),p.GetProperty("speed").GetDouble(),profile.GetProperty("delayTicks").GetInt32(),profile.GetProperty("brakeScale").GetDouble(),profile.GetProperty("gripDrop").GetBoolean(),scenario));
            }
            var a=Run(AtoMethod.AdaptivePredictive,1000,1.1,.55,0,null,1,.15,25,900,16,15,.8,true);
            var b=Run(AtoMethod.AdaptivePredictive,1000,1.1,.55,0,null,7,.15,25,900,16,15,.8,true);
            Require(a.finalState==b.finalState,"Delayed-observation frame replay mismatch");
            var old=JsonSerializer.Deserialize<List<Result>>(File.ReadAllText("research/ato/results/heldout-switch-v1/results.json"));
            foreach(var prior in old.GroupBy(r=>(r.method,r.dwellTicks)).Select(g=>g.First()))
            {
                var current=Run(Enum.Parse<AtoMethod>(prior.method),prior.seed,prior.mass,prior.grip,prior.grade,null,1,prior.penalty,prior.dwellTicks,prior.targetDistance,prior.targetSpeed);
                Require(current.finalState==prior.finalState,"Metrics changed a legacy trajectory");
            }
            File.Copy("research/ato/stress-protocol-v1.json",Path.Combine(dir,"protocol.json"));
        }
        bool invalid=false;try{new CabAtoController(AtoMethod.Predictive,1,0);}catch(ArgumentException){invalid=true;}
        Require(invalid,"Invalid task accepted");
        var controller=new CabAtoController(AtoMethod.Predictive,100,16);var o=new AtoObservation{Tick=1};controller.Decide(o);
        invalid=false;try{controller.Decide(o);}catch(ArgumentException){invalid=true;}Require(invalid,"Duplicate observation accepted");
        var overshoot=new CabAtoController(AtoMethod.Predictive,100,16).Decide(new AtoObservation{Tick=1,Position=101,Speed=4});
        Require(overshoot.Throttle==0&&overshoot.Brake==3,"Overshoot not braked");
        File.WriteAllText(Path.Combine(dir,"results.json"),JsonSerializer.Serialize(results,new JsonSerializerOptions{WriteIndented=true}));
        foreach(string source in new[]{"Assets/Scripts/Cab/CabSimulation.cs","Assets/Scripts/Cab/CabSimulationCodec.cs","Assets/Scripts/Cab/Research/CabAtoController.cs","research/ato/Program.cs","research/ato/AtoResearch.csproj"})
            File.Copy(source,Path.Combine(dir,Path.GetFileName(source)+".txt"),false);
        File.WriteAllText(Path.Combine(dir,"manifest.json"),JsonSerializer.Serialize(new{scope=stress?"Frozen finite stress cases; not field evidence":heldout?"Frozen held-out scenario grid; finite simulation conditions, not real-world evidence":"Development only. No held-out test or real-train claim.",protocolHash=stress?Hash("research/ato/stress-protocol-v1.json"):heldout?Hash("research/ato/heldout-protocol-v1.json"):null,utc=DateTime.UtcNow,sourceHash=Hash("Assets/Scripts/Cab/CabSimulation.cs"),controllerHash=Hash("Assets/Scripts/Cab/Research/CabAtoController.cs"),runtime=Environment.Version.ToString(),controlTicks=5,stepSeconds=.02,seedSet=results.Select(r=>r.seed).Distinct().ToArray(),rowCount=results.Count,stopSpeedMps=.01,stopHoldTicks=100,toleranceM=5,replayFrameGrouping=true,legacyMetricCompatibilityChecked=stress},new JsonSerializerOptions{WriteIndented=true}));
        Console.WriteLine("ATO development checks passed: replay, invalid inputs, overshoot guard.");
        foreach(var group in results.GroupBy(r=>$"{r.method}/penalty={r.penalty}/dwell={r.dwellTicks}"))Console.WriteLine($"{group.Key} n={group.Count()} stopped={group.Count(r=>r.stopped)} within5m={group.Count(r=>r.reachedTolerance)} meanAbsError={group.Average(r=>Math.Abs(r.errorM)):F2}m meanSwitches={group.Average(r=>r.switches):F1} maxDecision={group.Max(r=>r.decisionMaxMs):F2}ms");
    }
}
