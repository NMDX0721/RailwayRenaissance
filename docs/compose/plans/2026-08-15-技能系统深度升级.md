### 核心公式体系

#### 公式1：母技能等级计算（由子技能反馈驱动）

```
ParentSkill[i] = Σ(SubSkill[j] × w_j) / Σ(w_j) + PassiveGrowth
  - SubSkill[j] 属于母技能 i 的所有子技能
  - w_j = 1.0 + (SubSkill[j] 的岗位匹配度 × 0.5)
    - 司机岗位时，"行车·司机"子技能的 w = 1.5
  - PassiveGrowth = 0.01/天（缓慢自然增长）
```

#### 公式2：子技能日增长计算（核心公式）

```
SubSkillGain[k] = BaseGain × RoleMatch × Inertia × Fluctuation × ContextEvent
  × TalentModifier × IntelligenceModifier × MacroModifier

  BaseGain = WorldBaseline[system] × DifficultyMultiplier × (1 + GapBonus)
    GapBonus = max(0, (ParentSkill[i] - SubSkill[k]) / ParentSkill[i]) × 0.5
    // 子技能落后母技能越多，追赶越快（最大+50%）

  RoleMatch = 
    1.0  // 当前岗位匹配的子技能
    0.3  // 同系统其他子技能（关联技能）
    0.1  // 跨系统子技能（高门槛）
    0.3  // 跨系统子技能（低门槛）

  Inertia = 1.0 - (HistoricalAvg[k] - SubSkill[k]) × 0.15
    // 历史基线高于当前等级 → 惯性拖累
    // 历史基线低于当前等级 → 惯性助力
    // clamp 到 [0.5, 1.5]

  Fluctuation = 1.0 + Random(-Range, +Range)
    Range = 0.05 + EventSeverity × 0.03
    // 基准 5% 随机波动，事件加剧时最高可达 20%

  ContextEvent = 
    1.0      // 无事件
    2.0      // 老师傅传授
    1.5      // 培训课程
    2.5      // 外聘专家
    3.0      // 剧情事件
    0.5      // 负面事件（事故后信心受挫）

  TalentModifier = 0.5 + HiddenTalent / 100 * 0.5
    // 天赋高→学得快，天赋低→学得慢

  IntelligenceModifier = 0.5 + HiddenIntelligence / 100 * 0.5
    // 智力影响所有技能学习，独立于天赋

  MacroModifier = 0.8 + MacroEconomyIndex × 0.4
    // 经济景气时学习资源多，萧条时学习机会少
```

#### 公式3：子技能→母技能反馈

```
ParentSkillGain[i] = Σ(SubSkillChange[j]) / Count  × 0.3
  // 子技能提升总量的 30% 反馈注入母技能
```

#### 公式4：惯性沉积（历史基线更新）

```
H[t] = 0.4×S[t-1] + 0.3×S[t-2] + 0.2×S[t-3] + 0.1×S[t-4]
```

#### 公式5：波动区间生成

```
FluctuationRange = 0.05  // 基准 5%
  + (hasRecentAccident ? 0.05 : 0)  // 近期事故 +5%
  + (hasMentor ? -0.02 : 0)         // 有师傅指导 -2%
  + (isOverTime ? 0.03 : 0)         // 加班状态 +3%
  // 最终范围 [0.03, 0.20]
```

#### 公式6：社会对比效应（v3.0 新增）

```
SocialComparisonEffect[i] = Σ(max(0, Salary[j] - Salary[i]) / Salary[i] × 0.1)
  // 员工 i 发现员工 j 工资更高时，忠诚度下降
  // 差距越大，效果越明显
  // 受 HiddenPatience 调节：耐心高的人影响小
```

#### 公式7：非线性阈值效应（v3.0 新增）

```
非线性系数 = 
  1.0                                    // 正常区间
  × (fatigue > 80 ? 2.0 : 1.0)          // 疲劳>80事故概率翻倍
  × (loyalty < 30 ? 1.5 : 1.0)          // 忠诚<30离职概率+50%
  × (skillLevel > 80 ? 0.5 : 1.0)       // 技能>80成长速度减半（天花板效应）
  × (skillLevel < 20 ? 1.5 : 1.0)       // 技能<20成长加速（新手红利）
```

#### 公式8：系统间耦合（v3.0 新增 - 反馈回路）

```
// 票价 → 客流 → 收入 → 工资 → 忠诚 → 技能 → 效率 → 成本 → 票价
// 这是一个完整的反馈回路，每个环节都是上一环节的输出

票价 = f(运营成本, 市场, 政策)
  ↓
客流 = f(票价, 信任, 沙能, 服务)
  ↓
收入 = 客流 × 票价
  ↓
工资 = f(收入, 技能, 市场)
  ↓
忠诚度 = f(工资, 社会对比, 事件)
  ↓
技能成长 = f(忠诚度, 疲劳, 天赋)
  ↓
运营效率 = f(技能, 协同)
  ↓
运营成本 = f(运营效率, 维护, 燃料)
  ↓
票价 = f(运营成本, ...)  // 闭环
```

#### 公式9：外部宏观环境（v3.0 新增）

```
MacroModifier = 0.5 + MacroEconomyIndex × 0.5
  // 经济景气×1.0，经济萧条×0.5

SandEnergyTrendModifier = 1.0 + SandEnergyTrend × 0.3
  // 沙能上升时铁路技能成长受压制（-30%），沙能衰落时铁路技能更受重视（+30%）
```

```csharp
[System.Serializable]
public class SkillTreeNode
{
    public string systemName;         // "行车" / "机辆" / "线路" / "信号" / "站务"
    public float parentSkillLevel;    // 母技能等级 0-100
    public List<SubSkillData> subSkills;  // 子技能列表
    
    // 计算母技能等级（由子技能反馈驱动）
    public void RecalculateParentLevel()
    {
        float weightedSum = 0;
        float weightSum = 0;
        foreach (var sub in subSkills)
        {
            float w = 1.0f + sub.GetRoleMatchWeight() * 0.5f;
            weightedSum += sub.level * w;
            weightSum += w;
        }
        parentSkillLevel = (weightSum > 0) ? weightedSum / weightSum : 0;
        parentSkillLevel += 0.01f; // PassiveGrowth
    }
}

[System.Serializable]
public class SubSkillData
{
    public string skillName;          // "司机" / "调车员" / ...
    public string systemName;         // 所属母技能系统
    public float level;               // 子技能等级 0-100
    public bool isUnlocked;           // 是否已解锁（技能书）
    public float[] historicalAvg;     // 30天历史基线（环形缓冲区）
    public int historyIndex;          // 当前写入位置
    
    // 岗位匹配权重
    public float GetRoleMatchWeight(string currentRole)
    {
        if (currentRole == skillName) return 1.0f;
        // 同系统子技能
        if (IsSameSystem(currentRole)) return 0.3f;
        // 跨系统高门槛
        if (IsHighThreshold(currentRole)) return 0.1f;
        // 跨系统低门槛
        return 0.3f;
    }
}
```

```csharp
public void DailySkillUpdate(string currentRole, string currentSystem, float worldBaseline, float difficultyMult)
{
    foreach (var parent in skillTree)
    {
        foreach (var sub in parent.subSkills)
        {
            if (!sub.isUnlocked) continue;
            
            // 1. 岗位匹配系数
            float roleMatch = sub.GetRoleMatchWeight(currentRole);
            
            // 2. GapBonus：子技能落后母技能越多，追赶越快
            float gapBonus = 0;
            if (parent.parentSkillLevel > sub.level && parent.parentSkillLevel > 0)
            {
                gapBonus = (parent.parentSkillLevel - sub.level) / parent.parentSkillLevel * 0.5f;
            }
            
            // 3. 基准成长
            float baseGain = worldBaseline * difficultyMult * (1 + gapBonus);
            
            // 4. 惯性系数
            float historicalAvg = sub.historicalAvg.Average();
            float inertia = 1.0f - (historicalAvg - sub.level) * 0.15f;
            inertia = Mathf.Clamp(inertia, 0.5f, 1.5f);
            
            // 5. 波动区间
            float range = CalculateFluctuationRange();
            float fluctuation = 1.0f + UnityEngine.Random.Range(-range, range);
            
            // 6. 上下文事件
            float contextEvent = GetContextEventModifier(sub.skillName);
            
            // 7. 最终日增长
            float gain = baseGain * roleMatch * inertia * fluctuation * contextEvent;
            sub.level = Mathf.Clamp(sub.level + gain, 0, 100);
            
            // 8. 更新历史基线
            UpdateHistoricalAvg(sub, sub.level);
        }
        
        // 9. 子技能→母技能反馈
        parent.RecalculateParentLevel();
    }
}

private float CalculateFluctuationRange()
{
    float range = 0.05f;
    if (HasRecentAccident()) range += 0.05f;
    if (HasMentor()) range -= 0.02f;
    if (IsOverTime()) range += 0.03f;
    return Mathf.Clamp(range, 0.03f, 0.20f);
}

private void UpdateHistoricalAvg(SubSkillData sub, float currentLevel)
{
    sub.historicalAvg[sub.historyIndex % 30] = currentLevel;
    sub.historyIndex++;
}
```

```csharp
// 在 WorldGenerator 或 WorldInitializer 中
public float[] CalculateSkillBaselines(WorldSeedData seed)
{
    float[] baselines = new float[5];
    
    // 行车基准：受铁路里程、城市数量影响
    baselines[0] = 0.03f + seed.cities.Length * 0.002f;
    
    // 机辆基准：受车辆保有量、工业城市数量影响
    int industrialCities = seed.cities.Count(c => c.type == CityType.Industrial);
    baselines[1] = 0.02f + industrialCities * 0.003f;
    
    // 线路基准：受地形复杂度、线路长度影响
    baselines[2] = 0.02f + seed.railEdges.Length * 0.001f;
    
    // 信号基准：受技术城市数量、沙能渗透影响
    int techCities = seed.cities.Count(c => c.sandPenetrationBase > 0.3f);
    baselines[3] = 0.015f + techCities * 0.004f;
    
    // 站务基准：受人口总量、旅游城市数量影响
    float totalPop = seed.cities.Sum(c => c.population);
    baselines[4] = 0.03f + totalPop / 100000f * 0.001f;
    
    // clamp 到 [0.01, 0.10]
    for (int i = 0; i < 5; i++) baselines[i] = Mathf.Clamp(baselines[i], 0.01f, 0.10f);
    
    return baselines;
}
```

```csharp
public static string GetRankName(string systemName, float level)
{
    int rank = Mathf.FloorToInt(level / 14.3f); // 0-100 → 0-7级
    rank = Mathf.Clamp(rank, 0, 7);
    
    switch (systemName)
```

## 核心数学公式汇总（供实现参考）

```
1. ParentSkill[i] = Σ(SubSkill[j] × w_j) / Σ(w_j) + 0.01
   w_j = 1.0 + (匹配度 × 0.5)

2. ΔSubSkill = BaseGain × RoleMatch × Inertia × Fluctuation × ContextEvent
   BaseGain = WorldBaseline × DifficultyMult × (1 + GapBonus)
   GapBonus = max(0, (Parent - Sub) / Parent) × 0.5
   Inertia = clamp(1 - (H - S) × 0.15, 0.5, 1.5)

   Range = clamp(0.05 + EventSeverity × 0.03, 0.03, 0.20)

3. ParentSkillGain[i] = Σ(ΔSubSkill[j]) / Count × 0.3

4. H[t] = 0.4×S[t-1] + 0.3×S[t-2] + 0.2×S[t-3] + 0.1×S[t-4]
```

### Task 0: 分级波动计算函数家族（所有数值系统的基座）

**Files:** Create `Assets/Scripts/Core/FluctuationEngine.cs`

**Design:**
不是只有一个通用函数——而是设计**四级波动计算函数**，不同场景调用不同精度层级，避免过度设计或精度不足。

```
Level 1 - 简单波动：用于临时/次要数值
  输出 = 基准值 × (1 + 随机(-范围, +范围))
  适用：单次事件小修正、UI显示微调

Level 2 - 加权波动：用于日常结算
  输出 = 基准值 × (1 + Σ(因素×权重)) × 随机系数
  适用：技能成长、工资、客流日变化

Level 3 - 复合波动：用于核心经济/重大事件
  输出 = 基准值 × (1 + Σ(因素×权重)) × 随机系数 × 事件倍率 × 难度倍率
  适用：票价、补贴、事故概率、重大谈判

Level 4 - 黑箱波动：用于世界种子基准值生成
  输出 = 内部复杂模型(种子数据, 城市分布, 产业, 地形, 渗透率)
  适用：千里马创世核对世界基准值的计算（外部不可见）
```

**核心接口：**
```csharp
public class FluctuationEngine
{
    private WorldSeedData seed;
    private float difficultyMult;
    private System.Random worldRandom;  // 基于世界种子的确定性随机
    
    // L1: 简单波动
    public float Simple(float baseValue, float variance = 0.1f)
    {
        float offset = (float)(worldRandom.NextDouble() * 2 - 1) * variance;
        return baseValue * (1 + offset);
    }
    
    // L2: 加权波动
    public float Weighted(float baseValue, 
        Dictionary<string, float> factors, 
        Dictionary<string, float> weights,
        float variance = 0.1f)
    {
        float factorSum = 0;
        foreach (var kvp in factors)
            if (weights.ContainsKey(kvp.Key))
                factorSum += kvp.Value * weights[kvp.Key];
        float noise = (float)(worldRandom.NextDouble() * 2 - 1) * variance;
        return baseValue * (1 + factorSum) * (1 + noise);
    }
    
    // L3: 复合波动
    public float Compound(float baseValue,
        Dictionary<string, float> factors,
        Dictionary<string, float> weights,
        float eventMultiplier = 1.0f,
        float variance = 0.1f)
    {
        return Weighted(baseValue, factors, weights, variance) * eventMultiplier * difficultyMult;
    }
    
    // L4: 黑箱波动（千里马创世核内部使用）
    public float Blackbox(float[] seedData, Func<float[], float> internalModel)
    {
        return internalModel(seedData);  // 内部复杂模型，外部不可见
    }
}
```

**使用指南：**
| 场景 | 使用级别 | 原因 |
|------|---------|------|
| 技能日增长 | L2 加权 | 需多因素加权，但精度要求适中 |
| 工资计算 | L2 加权 | 同技能日增长 |
| 忠诚度变化 | L2 加权 | 加薪谈判中需考虑员工性格+历史关系 |
| 师徒收益 | L2 加权 | 等级差+匹配度 |
| 协同效应 | L2 加权 | 互补匹配度+双方等级 |
| 经营姿态 | L2 加权 | 姿态类型+城市倾向 |
| 票价/补贴/容量 | L3 复合 | 核心经济数值，需事件倍率+难度 |
| 事故概率 | L3 复合 | 重大安全事件，需事件倍率 |
| 世界基准值 | L4 黑箱 | 千里马创世核内部，不可见 |
