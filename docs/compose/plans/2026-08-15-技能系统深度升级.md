### 2.2 手动实现的函数

```csharp
// 3D Perlin Noise（基于 Mathf.PerlinNoise 组合）
public static float Perlin3D(float x, float y, float z) {
    float xy = Mathf.PerlinNoise(x, y);
    float xz = Mathf.PerlinNoise(x, z);
    float yz = Mathf.PerlinNoise(y, z);
    float yx = Mathf.PerlinNoise(y, x);
    float zx = Mathf.PerlinNoise(z, x);
    float zy = Mathf.PerlinNoise(z, y);
    return (xy + xz + yz + yx + zx + zy) / 6f;
}

// fBm（多层噪声叠加）
public static float fBm(float x, float y, int octaves = 4, float lacunarity = 2f, float persistence = 0.5f) {
    float value = 0f;
    float amplitude = 1f;
    float frequency = 1f;
    float maxValue = 0f;
    for (int i = 0; i < octaves; i++) {
        value += amplitude * Mathf.PerlinNoise(x * frequency, y * frequency);
        maxValue += amplitude;
        amplitude *= persistence;
        frequency *= lacunarity;
    }
    return value / maxValue;  // 归一化到 [0,1]
}
```

### 2.3 波动引擎 L1-L4

```csharp
public struct WeightedFactor {
    public string factorName;  // 影响因素名（用于匹配权重）
    public float value;        // 当前因素值
}

public class FluctuationEngine {
    private System.Random worldRandom;  // 基于世界种子的确定性随机
    private GlobalRules rules;          // 来自 WorldSeedData.globalRules
    private float difficultyMult;
    
    // 初始化：传入世界种子和 globalRules
    public void Initialize(WorldSeedData seed, float difficulty) { ... }
    
    // L1: 简单波动——临时/次要数值
    public float Simple(float baseValue, float variance = 0.1f) {
        float offset = (float)(worldRandom.NextDouble() * 2f - 1f) * variance;
        return baseValue * (1f + offset);
    }
    
    // L2: 加权波动——日常结算（技能成长/工资/忠诚度）
    public float Weighted(float baseValue, WeightedFactor[] factors, string formulaName) {
        // formulaName 从 globalRules.fluctuationWeights 中查找权重表
        float[] weights = rules.GetWeights(formulaName);
        float factorSum = 0f;
        for (int i = 0; i < factors.Length; i++) {
            factorSum += factors[i].value * weights[i];
        }
        // Perlin 噪声提供连续波动（基于时间+员工ID哈希）
        float noise = Mathf.PerlinNoise(Time.time * 0.01f, factors[0].value) * 2f - 1f;
        return baseValue * (1f + factorSum) * (1f + noise * rules.fluctuationIntensity);
    }
    
    // L3: 复合波动——核心经济/重大事件（票价/补贴/事故）
    public float Compound(float baseValue, WeightedFactor[] factors, string formulaName, float eventStrength) {
        float weighted = Weighted(baseValue, factors, formulaName);
        // Sigmoid 处理事件强度——小事件几乎无影响，大事件迅速增幅
        float sigmoid = 1f / (1f + Mathf.Exp(-eventStrength * rules.eventSigmoidSteepness));
        return weighted * (1f + sigmoid * rules.eventMaxMultiplier) * difficultyMult;
    }
    
    // L4: 黑箱——世界基准值生成（千里马创世核内部调用）
    public float Blackbox(string valueName, WorldSeedData seed) {
        return CalculateBaseline(valueName, seed);  // 内部 fBm 模型
    }
}
```

## 四、核心公式（全部系数来自 GlobalRules）

### 4.1 母技能等级

```
ParentSkill[i] = Σ(SubSkill[j] × w_j) / Σ(w_j) + rules.baseLearningRate × 0.01
    w_j = 1.0 + RoleMatch[j] × 0.5    // RoleMatch 来自 GlobalRules 权重表
```

### 4.2 子技能日增长

```
ΔSubSkill = rules.baseLearningRate × RoleMatch × Inertia × Fluctuation × ContextEvent
           × TalentMod × IntelMod × MacroMod

    GapBonus = max(0, (Parent - Sub) / Parent) × rules.gapBonusMaxRate
    Inertia = clamp(1 - (H - S) × rules.inertiaCoefficient, rules.inertiaClampMin, rules.inertiaClampMax)
    Fluctuation = 1 + PerlinNoise(time, 员工ID哈希) × rules.fluctuationIntensity
    ContextEvent = rules.eventMultipliers[eventType]  // 来自 GlobalRules
    TalentMod = 0.5 + crew.hiddenTalent / 100 × 0.5
    IntelMod = 0.5 + crew.hiddenIntelligence / 100 × 0.5
    MacroMod = 0.8 + seed.initialTrends.sandPenetration × 0.4
```

### 4.3 子技能→母技能反馈

```
ΔParentSkill = Σ(ΔSubSkill) / Count × rules.parentFeedbackRate
```

### 4.4 惯性沉积

```
H[t] = 0.4×S[t-1] + 0.3×S[t-2] + 0.2×S[t-3] + 0.1×S[t-4]
// 权重固定（指数衰减），但窗口长度可由 GlobalRules 配置
```

### 4.5 非线性阈值

```
实际倍率 = 1.0
    × (fatigue > rules.fatigueDangerThreshold ? rules.fatigueDangerMultiplier : 1.0)
    × (loyalty < rules.loyaltyDangerThreshold ? rules.loyaltyDangerMultiplier : 1.0)
    × (skill > rules.skillCeilingThreshold ? rules.skillCeilingMultiplier : 1.0)
    × (skill < rules.skillNewbieThreshold ? rules.skillNewbieMultiplier : 1.0)
```
